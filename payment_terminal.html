    This is the exact code running this page, split into the same sections the comments in the source use. Payment is the shared base class; CreditCardPayment, PayPalPayment, and BankTransferPayment each extend it with their own process(). PaymentProcessor holds a plain array of Payment objects and calls process() on each one — it never checks which subclass it's dealing with, yet the correct behavior always runs. The sections below walk through the base class, each payment type, the processor, and finally the plain DOM code that wires it all to the page.

    1. Payment — the abstract base class

    Every payment type needs the same basic shape: an id, an amount, a currency, a status, and a timestamp. Payment sets those fields up in its constructor, but it deliberately refuses to be used on its own — instantiating it directly throws, which mimics an "abstract" class in languages like Java. Its process() method also throws by default: it exists purely as a placeholder that every subclass is expected to override with real logic.

    let ticketCounter = 1000;

class Payment {
  constructor(amount, currency) {
    if (new.target === Payment) {
      throw new Error("Payment is abstract — instantiate a subclass instead.");
    }
    if (!(amount > 0)) {
      throw new Error("Payment amount must be positive.");
    }
    this.paymentId = "TXN" + (ticketCounter++);
    this.amount = amount;
    this.currency = currency;
    this.status = "PENDING";
    this.timestamp = new Date();
  }

  // Abstract method: every subclass MUST override this.
  // This is the single method PaymentProcessor calls — and the
  // reason dynamic polymorphism works: PaymentProcessor never
  // needs to know which override actually runs.
  process(log) {
    throw new Error("process() must be implemented by a subclass.");
  }

  // Overridable label — default, subclasses override it.
  get paymentType() {
    return "Generic Payment";
  }

  formattedTime() {
    return this.timestamp.toLocaleTimeString();
  }
}

    2. CreditCardPayment — overriding process() for cards

    CreditCardPayment extends Payment and adds the fields a card payment actually needs: card number, cardholder name, expiry, and CVV. Its process() override validates those details, masks the card number before it's ever logged or displayed, then simulates a network authorization check.

    class CreditCardPayment extends Payment {
  constructor(amount, currency, cardNumber, cardHolderName, expiry, cvv) {
    super(amount, currency);
    this.cardNumber = cardNumber;
    this.cardHolderName = cardHolderName;
    this.expiry = expiry;
    this.cvv = cvv;
  }

  get paymentType() {
    return "Credit Card";
  }

  maskedNumber() {
    const digits = this.cardNumber.replace(/\s/g, "");
    return "•••• •••• •••• " + digits.slice(-4);
  }

  // This OVERRIDE is what actually executes when PaymentProcessor
  // calls payment.process() on a CreditCardPayment instance.
  process(log) {
    log(`Cardholder  ${this.cardHolderName}`);
    log(`Card number ${this.maskedNumber()}`);

    const digits = this.cardNumber.replace(/\s/g, "");
    const validCard = digits.length >= 12 && /^\d{3}$/.test(this.cvv);
    if (!validCard) {
      this.status = "FAILED";
      log("Declined — invalid card details.", "fail");
      return false;
    }

    log("Contacting card network for authorization…");
    const authorized = this.amount <= 500000; // simulated network response
    if (authorized) {
      this.status = "SUCCESS";
      log(`Authorized — ${this.amount.toFixed(2)} ${this.currency} charged.`, "ok");
      return true;
    }
    this.status = "FAILED";
    log("Declined by issuer.", "fail");
    return false;
  }
}

    3. PayPalPayment — the same method, a different body

    PayPalPayment has nothing to do with card numbers — it stores an email and password instead — but it still implements process() with the exact same method signature as CreditCardPayment. That's the crux of polymorphism: one shared method name, independent implementations, each free to do whatever makes sense for that payment type.

    class PayPalPayment extends Payment {
  constructor(amount, currency, email, password) {
    super(amount, currency);
    this.email = email;
    this.password = password;
  }

  get paymentType() {
    return "PayPal";
  }

  // Same method NAME as CreditCardPayment.process(), completely
  // different BODY. That's polymorphism: one shared interface,
  // many independent implementations.
  process(log) {
    log(`PayPal account  ${this.email}`);

    const validAccount = this.email.includes("@") && this.password.length >= 4;
    if (!validAccount) {
      this.status = "FAILED";
      log("Declined — invalid PayPal credentials.", "fail");
      return false;
    }

    log("Redirecting to PayPal for authentication…");
    const confirmed = this.amount <= 250000; // simulated balance check
    if (confirmed) {
      this.status = "SUCCESS";
      log(`Confirmed — ${this.amount.toFixed(2)} ${this.currency} sent via PayPal.`, "ok");
      return true;
    }
    this.status = "FAILED";
    log("Declined — insufficient PayPal balance.", "fail");
    return false;
  }
}

    4. BankTransferPayment — a third, independent implementation

    Bank transfers need bank name, account number, and routing code, and settlement works differently from a card or PayPal charge — it simulates an interbank settlement step none of the other types have. Same process() interface, yet again a completely different body.

    class BankTransferPayment extends Payment {
  constructor(amount, currency, bankName, accountNumber, routingCode) {
    super(amount, currency);
    this.bankName = bankName;
    this.accountNumber = accountNumber;
    this.routingCode = routingCode;
  }

  get paymentType() {
    return "Bank Transfer";
  }

  maskedAccount() {
    return "XXXX" + this.accountNumber.slice(-4);
  }

  process(log) {
    log(`Bank        ${this.bankName}`);
    log(`Account     ${this.maskedAccount()}`);
    log(`Routing     ${this.routingCode}`);

    const validDetails = this.accountNumber.length >= 6 && this.routingCode.length > 0;
    if (!validDetails) {
      this.status = "FAILED";
      log("Declined — invalid bank details.", "fail");
      return false;
    }

    log("Initiating interbank settlement…");
    const settled = this.amount <= 1000000; // simulated settlement response
    if (settled) {
      this.status = "SUCCESS";
      log(`Settled — ${this.amount.toFixed(2)} ${this.currency} transferred.`, "ok");
      return true;
    }
    this.status = "FAILED";
    log("Rejected by receiving bank.", "fail");
    return false;
  }
}

    5. PaymentProcessor — where the polymorphism is actually used

    This is the class that matters most for the lesson. Its queue is just a plain array — it can hold a mix of CreditCardPayment, PayPalPayment, and BankTransferPayment objects side by side, with no tagging or type-checking. processAll() loops over that array and calls payment.process(log) on each entry. JavaScript resolves that call using the object's real class at runtime, so the correct override always runs — PaymentProcessor itself never asks "what kind of payment is this?" anywhere in its code.

    class PaymentProcessor {
  constructor() {
    this.queue = [];
    this.history = [];
  }

  addPayment(payment) {
    this.queue.push(payment);
  }

  // Dynamic dispatch happens on the next line: `payment.process(...)`
  // resolves to a different function body depending on what `payment`
  // really is, decided at runtime — not by anything written here.
  processAll(onEntry) {
    const results = [];
    for (const payment of this.queue) {
      const lines = [];
      const log = (text, kind) => lines.push({ text, kind: kind || "info" });
      const success = payment.process(log);
      this.history.push(payment);
      results.push({ payment, lines, success });
      if (onEntry) onEntry(payment, lines, success);
    }
    this.queue = [];
    return results;
  }

  totalSuccessful() {
    return this.history
      .filter((p) => p.status === "SUCCESS")
      .reduce((sum, p) => sum + p.amount, 0);
  }
}

    6. Wiring the interface

    This block connects the classes above to the actual page. It creates the single PaymentProcessor instance the whole demo shares, wires up the credit card / PayPal / bank transfer tabs, grabs references to the DOM elements the rest of the script needs, and defines the render functions that keep the queue list, the receipt feed, and the running totals in sync with the processor's state. showError() handles validation failures by flashing the red banner for a few seconds.

    const processor = new PaymentProcessor();

const tabs = document.querySelectorAll(".tab");
const panels = document.querySelectorAll(".form-panel");
tabs.forEach((tab) => {
  tab.addEventListener("click", () => {
    tabs.forEach((t) => t.classList.remove("active"));
    panels.forEach((p) => p.classList.remove("active"));
    tab.classList.add("active");
    document.getElementById("panel-" + tab.dataset.type).classList.add("active");
  });
});

const queueList = document.getElementById("queue-list");
const queueEmpty = document.getElementById("queue-empty");
const processBtn = document.getElementById("process-btn");
const receiptFeed = document.getElementById("receipt-feed");
const receiptEmpty = document.getElementById("receipt-empty");
const totalValue = document.getElementById("total-value");
const historyCount = document.getElementById("history-count");
const errorBanner = document.getElementById("error-banner");

function showError(message) {
  errorBanner.textContent = message;
  errorBanner.classList.add("visible");
  clearTimeout(showError._t);
  showError._t = setTimeout(() => errorBanner.classList.remove("visible"), 3200);
}

function renderQueue() {
  queueList.innerHTML = "";
  if (processor.queue.length === 0) {
    queueEmpty.style.display = "block";
    processBtn.disabled = true;
  } else {
    queueEmpty.style.display = "none";
    processBtn.disabled = false;
    processor.queue.forEach((p) => {
      const row = document.createElement("li");
      row.className = "queue-row type-" + p.paymentType.toLowerCase().replace(/\s/g, "-");
      row.innerHTML = `
        <span class="queue-type">${p.paymentType}</span>
        <span class="queue-id">${p.paymentId}</span>
        <span class="queue-amount">${p.amount.toFixed(2)} ${p.currency}</span>
      `;
      queueList.appendChild(row);
    });
  }
  processBtn.textContent = `Process all (${processor.queue.length})`;
}

function buildTicket(payment, lines, success) {
  const ticket = document.createElement("div");
  ticket.className = "ticket " + (success ? "ticket-ok" : "ticket-fail");

  const logLines = lines
    .map((l) => `<div class="log-line log-${l.kind}">${l.text}</div>`)
    .join("");

  ticket.innerHTML = `
    <div class="ticket-head">
      <span class="ticket-type">${payment.paymentType}</span>
      <span class="ticket-id">${payment.paymentId}</span>
    </div>
    <div class="ticket-time">${payment.formattedTime()}</div>
    <div class="ticket-log">${logLines}</div>
    <div class="ticket-stamp">${success ? "SUCCESS" : "DECLINED"}</div>
  `;
  return ticket;
}

function updateTotals() {
  totalValue.textContent = processor.totalSuccessful().toFixed(2);
  historyCount.textContent = processor.history.length;
}

    6a. Sample data

function fillSample(type) {
  const s = samples[type];
  Object.entries(s).forEach(([key, val]) => {
    const el = document.getElementById(type + "-" + key);
    if (el) el.value = val;

    6c. Processing the queue

    When "Process all" is clicked, processAll() runs every queued payment's own process() override and returns the results. The click handler then prints one receipt ticket per result with a short stagger, so it reads like a printer rather than everything appearing at once.

    // PaymentProcessor.processAll() takes a callback that fires once per
// payment, right after that payment's own process() override has run.
// We use it to print tickets with a short stagger, so it reads like a
// receipt printing out rather than all results appearing at once.
processBtn.onclick = () => {
  if (processor.queue.length === 0) return;
  processBtn.disabled = true;
  receiptEmpty.style.display = "none";

  const entries = [];
  processor.processAll((payment, lines, success) => entries.push({ payment, lines, success }));

  entries.forEach((entry, i) => {
    setTimeout(() => {
      const ticket = buildTicket(entry.payment, entry.lines, entry.success);
      receiptFeed.prepend(ticket);
      updateTotals();
      if (i === entries.length - 1) {
        renderQueue();
      }
    }, i * 260);
  });
};
